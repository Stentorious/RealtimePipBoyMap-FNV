; Update screen on Chameleon effect change
SetOnActorValueChangeEventHandler 1 (begin function {int iAVCode, float fPrevious, float fNew}
	if eval Goo1.AuxVarGetFlt "*PipMap_Anim" == 0
		return
	endif
	CallAfterFrames 0 (begin function {}
		Player.SetNifBlockFlag "pipboyscreen:0" 0 0
		if eval Player.GetAV Chameleon > 0
			Player.SetMatrPropVal "pipboyscreen:0" 7 0.99
		else
			Player.SetMatrPropVal "pipboyscreen:0" 7 1
		endif
	end)
end) 0 49

; Update Pip-Boy radiation gauge
SetOnActorValueChangeEventHandler 1 (begin function {int iAVCode, float fPrevious, float fNew}
	if eval Goo1.AuxVarGetFlt "*PipMap_Anim" == 0 || ModelHasBlock Player "RadNeedle" == 0
		return
	endif
	Player.GetNifRotAlt "RadNeedle" (float fTemp1) (float fTemp2) (float fTemp3) 2 2
	fTemp3 = (((Player.GetAV RadiationRads) - 500) / 500) * 90
	Player.SetNifRot "RadNeedle" (fTemp1) (fTemp2) (fTemp3) 2 2
end) 0 54
